### 设计模式

#### 1. 管理器模式 (Manager Pattern)

`AnnotationManager`作为中央管理器，协调所有组件：

```typescript
export class AnnotationManager implements IDisposable {
    private _document: IDocument;
    private _annotations = new Map<string, AnnotationNode>();
    private _history = new AnnotationHistory();
    private _validator: IAnnotationValidator;
    
    // 统一的接口管理所有标注操作
    createAnnotation(type: MachiningFeatureType, name?: string): AnnotationNode
    deleteAnnotation(annotationId: string): boolean
    selectFaces(faceIds: number[], append = false): void
}
```

#### 2. 策略模式 (Strategy Pattern)

验证器和导出器使用策略模式：

```typescript
// 验证器策略
export interface IAnnotationValidator {
    validateAnnotation(annotation: Annotation): ValidationResult;
    validateFaces(annotation: Annotation, faceIds: number[]): ValidationResult;
}

// 导出器策略  
export abstract class BaseAnnotationExporter {
    abstract export(annotations: AnnotationNode[]): any;
}

// 使用策略
const manager = new AnnotationManager(document, new CustomValidator());
const exporter = new AAGNetExporter();
```

#### 3. 观察者模式 (Observer Pattern)

事件系统实现组件间解耦：

```typescript
export class AnnotationManager {
    private _onAnnotationCreated: ((node: AnnotationNode) => void)[] = [];
    private _onAnnotationDeleted: ((node: AnnotationNode) => void)[] = [];
    
    onAnnotationCreated(callback: (node: AnnotationNode) => void): void {
        this._onAnnotationCreated.push(callback);
    }
    
    // 触发事件
    private triggerAnnotationCreated(node: AnnotationNode): void {
        this._onAnnotationCreated.forEach(callback => callback(node));
    }
}
```

#### 4. 命令模式 (Command Pattern)

与chili3d命令系统集成：

```typescript
@command({
    key: "annotation.start",
    icon: "icon-play",
})
export class StartAnnotationCommandRegistered implements ICommand {
    async execute(application: IApplication): Promise<void> {
        // 命令执行逻辑
    }
}
```

#### 5. 装饰器模式 (Decorator Pattern)

chili3d属性系统集成：

```typescript
export class AnnotationNode extends Node {
    @Serializer.serialze()
    @Property.define("annotation.featureType" as any)
    get featureType(): MachiningFeatureType {
        return this._annotation.type;
    }
}
```

### 架构原则

#### 1. 单一职责原则
每个类专注单一功能：
- `Annotation`: 数据模型
- `AnnotationManager`: 管理协调
- `AnnotationPanel`: 用户界面
- `FaceSelectionHandler`: 交互处理

#### 2. 开闭原则  
通过接口扩展功能：
- `IAnnotationValidator`: 支持自定义验证
- `BaseAnnotationExporter`: 支持新导出格式

#### 3. 依赖倒置原则
依赖抽象而非具体实现：
- 管理器依赖验证器接口
- 导出系统依赖基类抽象

```typescript
private validateFeatureTypeConstraints(annotation: Annotation, result: ValidationResult): void {
```

### 开发自定义验证器

#### 实现验证器接口

```typescript
// src/validators/myCustomValidator.ts
export class MyCustomValidator implements IAnnotationValidator {
    validateAnnotation(annotation: Annotation): ValidationResult {
        const result: ValidationResult = {
            isValid: true,
            errors: [],
            warnings: []
        };

        // 自定义验证逻辑
        if (annotation.name.includes("forbidden")) {
            result.errors.push("Annotation name contains forbidden word");
            result.isValid = false;
        }

        // 特定特征类型的自定义验证
        if (annotation.type === MachiningFeatureType.CustomFeature) {
            result.warnings.push("Custom feature requires manual review");
        }

        return result;
    }

    validateFaces(annotation: Annotation, faceIds: number[]): ValidationResult {
        const result: ValidationResult = {
            isValid: true,
            errors: [],
            warnings: []
        };

        // 自定义面验证逻辑
        for (const faceId of faceIds) {
            if (faceId > 1000) {
                result.warnings.push(`Face ID ${faceId} is unusually large`);
            }
        }

        return result;
    }

    validateFeatureType(annotation: Annotation): ValidationResult {
        // 特征类型验证逻辑
        return { isValid: true, errors: [], warnings: [] };
    }

    validateCompleteness(annotation: Annotation): ValidationResult {
        // 完整性验证逻辑  
        return { isValid: true, errors: [], warnings: [] };
    }
}
```

### 开发自定义导出器

#### 继承基础导出器

```typescript
// src/exporters/myCustomExporter.ts
export class MyCustomExporter extends BaseAnnotationExporter {
    constructor(config: ExportConfig = {}) {
        super({ ...config, format: "my-custom" });
    }

    export(annotations: AnnotationNode[], modelFileName: string, totalFaceCount: number): any {
        // 构建自定义格式数据
        const exportData = {
            format: "MyCustomFormat",
            version: "1.0.0",
            timestamp: new Date().toISOString(),
            model: modelFileName,
            features: annotations.map(node => this.convertAnnotation(node))
        };

        return exportData;
    }

    validate(exportedData: any): { isValid: boolean; errors: string[] } {
        const errors: string[] = [];

        // 验证导出数据格式
        if (!exportedData.format || exportedData.format !== "MyCustomFormat") {
            errors.push("Invalid format identifier");
        }

        if (!Array.isArray(exportedData.features)) {
            errors.push("Features must be an array");
        }

        return { isValid: errors.length === 0, errors };
    }

    protected getFormatName(): string {
        return "My Custom Format";
    }

    private convertAnnotation(node: AnnotationNode): any {
        return {
            id: node.annotation.id,
            name: node.annotation.name,
            type: node.featureType,
            typeName: node.featureName,
            faces: node.faces,
            confidence: node.confidence,
            validated: node.validated,
            geometricParams: node.geometricParams,
            toleranceSpec: node.toleranceSpec,
            metadata: node.annotation.metadata
        };
    }
}
```

```typescript
// src/commands/myCustomCommand.ts
@command({
    key: "annotation.custom.analyze",
    icon: "icon-analytics",
})
export class CustomAnalyzeCommand implements ICommand {
    async execute(application: IApplication): Promise<void> {
        const document = application.activeView?.document;
        if (!document) {
            alert("No active document found");
            return;
        }

        try {
            // 获取标注管理器
            const annotationManager = (document as any)._annotationManager;
            if (!annotationManager) {
                alert("Annotation system not started");
                return;
            }

            // 执行自定义分析
            const results = await this.analyzeAnnotations(annotationManager.annotations);
            
            // 显示结果
            this.displayResults(results);
            
        } catch (error) {
            console.error("Custom analysis failed:", error);
            alert(`Analysis failed: ${error}`);
        }
    }

    private async analyzeAnnotations(annotations: AnnotationNode[]): Promise<any> {
        // 自定义分析逻辑
        return {
            totalAnnotations: annotations.length,
            featureTypes: this.getFeatureTypeDistribution(annotations),
            coverage: this.calculateCoverage(annotations),
            quality: this.assessQuality(annotations)
        };
    }

    private getFeatureTypeDistribution(annotations: AnnotationNode[]): Record<string, number> {
        const distribution: Record<string, number> = {};
        
        annotations.forEach(node => {
            const typeName = node.featureName;
            distribution[typeName] = (distribution[typeName] || 0) + 1;
        });
        
        return distribution;
    }
}
```

#### 事件系统优化

```typescript
export class AnnotationManager {
    private _eventQueue: Array<() => void> = [];
    private _isProcessingEvents = false;
    
    private triggerEvent(eventFn: () => void): void {
        this._eventQueue.push(eventFn);
        
        if (!this._isProcessingEvents) {
            this.processEventQueue();
        }
    }
    
    private processEventQueue(): void {
        this._isProcessingEvents = true;
        
        // 使用setTimeout确保不阻塞主线程
        setTimeout(() => {
            while (this._eventQueue.length > 0) {
                const eventFn = this._eventQueue.shift();
                if (eventFn) {
                    try {
                        eventFn();
                    } catch (error) {
                        console.error('Event processing error:', error);
                    }
                }
            }
            
            this._isProcessingEvents = false;
        }, 0);
    }
}
```

```typescript
class OptimizedFaceSelection {
    private processInBatches<T>(items: T[], batchSize: number, processor: (batch: T[]) => void): Promise<void> {
        return new Promise((resolve) => {
            let index = 0;
            
            const processBatch = () => {
                const batch = items.slice(index, index + batchSize);
                if (batch.length === 0) {
                    resolve();
                    return;
                }
                
                processor(batch);
                index += batchSize;
                
                // 下一帧处理下一批
                requestAnimationFrame(processBatch);
            };
            
            processBatch();
        });
    }
    
    async selectLargeFaceSet(faceIds: number[]): Promise<void> {
        await this.processInBatches(faceIds, 100, (batch) => {
            // 处理每批面
            batch.forEach(id => this.selectFace(id));
        });
    }
}
```
